4.3 Button（按钮）

Button 是最常用的交互控件：四态外观（normal/hover/pressed/disabled）、标题文本（内部 Label）、可选 LuotiAni 动画。继承链：Button → ControlImpl → Control。

4.3.1 控件的创建

// C++
Button(Control* parent, SRect rect, float xScale = 1.0f, float yScale = 1.0f);
ButtonBuilder(parent, SRect(20, 20, 120, 36)).setCaption("OK").setOnClick(cb).build();

// C ABI
UIControlHandle btn = UICornerstone_CreateButton(g_inst, "OK", 20, 20, 120, 36, 1.0f, 1.0f);

// JSON
{ "type": "button", "id": "submitBtn", "rect": { "x": 280, "y": 140, "w": 100, "h": 35 },
  "caption": "Submit", "events": { "onClick": "onSubmitClick" } }

}

4.3.5 C++ Binding 样例

auto button = ui->CreateButton("Toggle", 20, 80, 120, 36);
int count = 0;
button.SetCallback(PropertyNames::kEventClick, [&](const Event&) {
    bool vis = label.GetBool(PropertyNames::kVisible);
    label.SetBool(PropertyNames::kVisible, !vis);
    label.SetString(PropertyNames::kCaption, "Clicked " + std::to_string(++count));
});
// 动态换文案 / 禁用
button.SetString(PropertyNames::kCaption, "新标题");
button.SetBool(PropertyNames::kEnabled, false);
给按钮挂动画：JSON "luotiAni": "assets/animations/rotateBtn/rotateBtn.jsonc" 或 C++ btn->setLuotiAni(shared_ptr<LuotiAni>)， 或直接用动画按钮工厂 UICornerstone_CreateAnimatedButton / ui->CreateAnimatedButton(jsoncPath, x, y, w, h, xScale, yScale) （scale 作用于按钮本身，内嵌动画恒 1.0，见 5.5 控件缩放）。

4.3.6 C ABI 调用样例

// 创建（见 4.3.1）：工厂 + 初始参数
UIControlHandle btn = UICornerstone_CreateButton(g_inst, "OK", 20, 20, 120, 36, 1.0f, 1.0f);

// 属性：泛型 Set*，键名见「控件的属性」表
UICornerstone_SetString(g_inst, btn, "caption", "Submit");
UICornerstone_SetStateColor(g_inst, btn, "background",
    (UIStateColor){ {0x4A,0x90,0xD9,0xFF}, {0x5B,0xA0,0xE9,0xFF}, {0x3A,0x80,0xC9,0xFF}, {0x80,0x80,0x80,0xFF} });

// 回调：事件名见「控件的回调」表
static void onBtnClick(UIControlHandle ctl, const UIEventData* ev, void* ud) {
    /* 点击处理 */
}
UICornerstone_SetCallback(g_inst, btn, "click", onBtnClick, NULL);

// 通用操作
UICornerstone_SetRect(g_inst, btn, 20, 20, 140, 40);
UICornerstone_DestroyControl(g_inst, btn);   // 不再需要时销毁
