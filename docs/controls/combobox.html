4.6 ComboBox（下拉选择框）

ComboBox 是支持只读 / 可编辑双模式的下拉选择控件：闭合时显示当前选中项，展开时下拉列表（超出显示滚动条），支持循环切换（↑/↓/滚轮）。继承链：ComboBox → EditBox → ControlImpl → Control。

缩放联动：主框文本与下拉列表项共用同一字体（继承 EditBox），置于缩放画布（fit / stretch，见 5.6 视口画布缩放）时字号随复合缩放系数重建。

4.6.1 控件的创建

// C++
ComboBox(Control* parent, SRect rect, float xScale = 1.0f, float yScale = 1.0f);
ComboBoxBuilder(parent, SRect(20, 56, 300, 32))
    .setItems({ {"Beijing", "beijing"}, {"Shanghai", "shanghai"} })
    .setSelectedIndex(0).build();

// C ABI
UIControlHandle combo = UICornerstone_CreateComboBox(g_inst, 20, 56, 300, 32, 1.0f, 1.0f);

// JSON
{ "id": "comboMain", "type": "combo-box",
  "rect": { "x": 20, "y": 56, "w": 300, "h": 32 },
  "editable": false, "placeholder": "Select a city...",
  "items": [
    { "label": "Beijing", "value": "beijing" },
    { "label": "Wuhan", "value": "wuhan", "disabled": true }
  ],
  "selected-index": 0, "max-visible-items": 6, "cycle-enabled": true,
  "events": { "onSelectionChanged": "onSelectionChanged" } }

4.6.5 C++ Binding 样例

Control combo = ui->CreateComboBox(20, 56, 300, 32);
// items 用 JSON 数组字符串设置
combo.SetString(PropertyNames::kItems, R"([{"label":"Beijing","value":"beijing"},
                             {"label":"Shanghai","value":"shanghai"}])");
combo.SetBool(PropertyNames::kEditable, false);
combo.SetCallback(PropertyNames::kEventSelectionChanged, [](const Event& e) {
    printf("selected index=%d value=%s\n", e.GetSelectedIndex(), e.GetSelectedValue().c_str());
});

// C++ 核心库
vector items = { {"Option A", "val_a"}, {"Option B", "val_b"} };
auto combo2 = ComboBoxBuilder(nullptr, SRect(CX, 40, CW, HH))
    .setItems(items).setSelectedIndex(0).setPlaceholder("Select an option...")
    .setOnSelectionChanged([](shared_ptr, int idx, const string& value) {
        cout << "index=" << idx << " value=" << value << endl;
    })
    .build();
BENCH->addControl(combo2);

4.6.6 C ABI 调用样例

UIControlHandle box = UICornerstone_CreateComboBox(g_inst, 20, 20, 160, 26, 1.0f, 1.0f);

UICornerstone_SetString(g_inst, box, "items", "Red,Green,Blue");  // 逗号分隔
UICornerstone_SetInt(g_inst, box, "selected-index", 1);
UICornerstone_SetBool(g_inst, box, "editable", 0);

static void onSelectionChanged(UIControlHandle ctl, const UIEventData* ev, void* ud) {
    /* ev->data.selection.idx = 选中下标；ev->data.selection.val = 选中项文本 */
}
UICornerstone_SetCallback(g_inst, box, "selection-changed", onSelectionChanged, NULL);
